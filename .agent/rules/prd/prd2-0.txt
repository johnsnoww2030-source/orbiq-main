۴.۱ ماژول نرخ ارز
text

┌─────────────────────────────────────────────────────────────────┐
│                                                                 │
│   FR-2.1: ثبت نرخ ارز                                          │
│   ─────────────────────                                         │
│   • نرخ ارز جدید فقط اضافه (Append) می‌شود                     │
│   • ویرایش و حذف نرخ‌های قبلی ممنوع است                        │
│   • هر نرخ شامل: ارز، مقدار، زمان، منبع، ثبت‌کننده            │
│   • منابع مجاز: manual, api_bonbast, api_tgju, correction      │
│   • confidence score برای هر نرخ ذخیره می‌شود                  │
│                                                                 │
│   FR-2.2: واکشی نرخ فعلی                                       │
│   ──────────────────────                                        │
│   • آخرین نرخ ثبت شده برای هر ارز برگردانده می‌شود            │
│   • کش ۵ دقیقه‌ای برای کاهش query                               │
│                                                                 │
│   FR-2.3: واکشی نرخ تاریخی                                     │
│   ─────────────────────────                                     │
│   • جستجوی نرخ در تاریخ مشخص                                   │
│   • اگر نرخ دقیق نبود → نزدیک‌ترین نرخ قبل                     │
│   • اگر هیچ نرخی نبود → اولین نرخ موجود                        │
│                                                                 │
│   FR-2.4: تاریخچه نرخ                                          │
│   ────────────────────                                          │
│   • لیست نرخ‌ها با فیلتر زمانی                                  │
│   • محاسبه درصد تغییر                                          │
│   • داده برای نمودار                                           │
│                                                                 │

text

┌─────────────────────────────────────────────────────────────────┐
│                                                                 │
│   FR-2.5: تنظیمات سود                                          │
│   ────────────────────                                          │
│   • ذخیره minProfitMargin, defaultProfitMargin, maxProfitMargin│
│   • اعتبارسنجی: min < default < max                            │
│   • محدوده: ۰٪ تا ۲۰۰٪                                         │
│   • پیش‌فرض: ۲۰٪ / ۳۰٪ / ۵۰٪                                   │
│                                                                 │
│   FR-2.6: محاسبه سه قیمت                                       │
│   ──────────────────────                                        │
│   • minPrice = costPrice × (1 + minProfitMargin/100)           │
│   • sellingPrice = costPrice × (1 + defaultProfitMargin/100)   │
│   • maxPrice = costPrice × (1 + maxProfitMargin/100)           │
│   • گرد کردن به نزدیک‌ترین ۱۰۰۰ تومان                          │
│                                                                 │
│   FR-2.7: اعتبارسنجی قیمت فروش                                 │
│   ─────────────────────────────                                 │
│   • اگر قیمت < minPrice → هشدار قرمز                           │
│   • اگر قیمت > maxPrice → هشدار زرد                            │
│   • نیاز به تایید مدیر برای فروش زیر کف                        │
│                                                                 │
└─────────────────────────────────────────────────────────────────┘
۴.۳ ماژول سود
text

┌─────────────────────────────────────────────────────────────────┐
│                                                                 │
│   FR-2.8: محاسبه سود فروش                                      │
│   ─────────────────────────                                     │
│   • profitIRR = sellingPrice - costPrice (ذخیره می‌شود)        │
│   • exchangeRateAtSale = نرخ فعلی (ذخیره می‌شود)              │
│   • profitUSD = profitIRR / exchangeRateAtSale (Computed)     │
│                                                                 │
│   FR-2.9: گزارش سود تاریخی                                     │
│   ──────────────────────────                                    │
│   • جمع profitIRR در بازه زمانی                                │
│   • جمع profitUSD (محاسبه با نرخ زمان هر فروش)                 │
│   • نرخ مؤثر = SUM(profitIRR) / SUM(profitUSD)                 │
│                                                                 │
│   FR-2.10: گزارش ارزش امروز                                    │
│   ────────────────────────────                                  │
│   • ارزش امروز = SUM(profitIRR) / نرخ فعلی                     │
│   • کاهش ارزش = (تاریخی - امروز) / تاریخی × ۱۰۰               │
│                                                                 │
└─────────────────────────────────────────────────────────────────┘

CREATE TABLE exchange_rate_events (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    currency_code TEXT NOT NULL,           -- USD, EUR, AED
    rate REAL NOT NULL,                    -- نرخ به ریال
    recorded_at TEXT NOT NULL,             -- ISO 8601 DateTime
    source TEXT NOT NULL,                  -- manual, api_bonbast, correction
    confidence REAL DEFAULT 1.0,           -- 0.0 to 1.0
    notes TEXT,                            -- توضیحات اختیاری
    recorded_by_uuid TEXT NOT NULL,        -- کاربر ثبت‌کننده
    
    -- Constraints
    CHECK (rate > 0),
    CHECK (confidence >= 0 AND confidence <= 1),
    CHECK (source IN ('manual', 'api_bonbast', 'api_tgju', 
                      'market_avg', 'correction'))
);

-- Index برای جستجوی سریع
CREATE INDEX idx_exchange_rate_currency_date 
    ON exchange_rate_events(currency_code, recorded_at DESC);

// ═══════════════════════════════════════════════════════════════

import 'package:equatable/equatable.dart';

enum RateSource { manual, apiBonbast, apiTgju, marketAvg, correction }

class ExchangeRate extends Equatable {
  final int id;
  final String currencyCode;
  final double rate;
  final DateTime recordedAt;
  final RateSource source;
  final double confidence;
  final String? notes;
  final String recordedByUuid;

  const ExchangeRate({

    required this.currencyCode,
    required this.rate,
    required this.recordedAt,
    required this.source,
    this.confidence = 1.0,
    this.notes,
    required this.recordedByUuid,
  });

  @override
  List<Object?> get props => [
    id, currencyCode, rate, recordedAt, 
    source, confidence, notes, recordedByUuid
  ];
}

// ═══════════════════════════════════════════════════════════════
// lib/features/pricing/domain/entities/price_range.dart
// ═══════════════════════════════════════════════════════════════

class PriceRange extends Equatable {
  final double minPrice;      // کف قیمت
  final double sellingPrice;  // قیمت فروش پیشنهادی
  final double maxPrice;      // سقف قیمت
  final double costPrice;     // قیمت تمام‌شده
  final double exchangeRate;  // نرخ ارز استفاده شده

  const PriceRange({
    required this.minPrice,
    required this.sellingPrice,
    required this.maxPrice,
    required this.costPrice,
    required this.exchangeRate,
  });

  /// درصد سود در قیمت فعلی
  double get currentMargin => (sellingPrice - costPrice) / costPrice;

  /// سود ریالی
  double get profitIRR => sellingPrice - costPrice;

  /// سود دلاری (با نرخ فعلی)
  double get profitUSD => profitIRR / exchangeRate;

  @override
  List<Object> get props => [
    minPrice, sellingPrice, maxPrice, costPrice, exchangeRate
  ];
}

// ═══════════════════════════════════════════════════════════════
// lib/features/pricing/domain/entities/profit.dart
// ═══════════════════════════════════════════════════════════════

/// سود عملیاتی: تفاوت قیمت فروش و قیمت تمام‌شده
class OperationalProfit extends Equatable {
  final double amountIRR;
  final double exchangeRateAtSale;
  final DateTime soldAt;

  const OperationalProfit({
    required this.amountIRR,
    required this.exchangeRateAtSale,
    required this.soldAt,
  });

  /// سود دلاری با نرخ زمان فروش (تاریخی)
  double get historicalUSD => amountIRR / exchangeRateAtSale;

  /// سود دلاری با نرخ دیگر (محاسباتی)
  double toUSD(double currentRate) => amountIRR / currentRate;

  @override
  List<Object> get props => [amountIRR, exchangeRateAtSale, soldAt];
}

/// تاثیر تغییر ارز (برای نمایش)
class FXRevaluationImpact extends Equatable {
  final double historicalValue;  // ارزش با نرخ قدیم
  final double currentValue;     // ارزش با نرخ امروز
  
  const FXRevaluationImpact({
    required this.historicalValue,
    required this.currentValue,
  });

  double get impact => currentValue - historicalValue;
  double get impactPercent => 
    historicalValue != 0 ? (impact / historicalValue) * 100 : 0;

  @override
  List<Object> get props => [historicalValue, currentValue];
}

// ═══════════════════════════════════════════════════════════════
// lib/features/currency/domain/repositories/exchange_rate_repository.dart
// ═══════════════════════════════════════════════════════════════

import 'package:dartz/dartz.dart';

abstract class ExchangeRateRepository {
  /// دریافت آخرین نرخ یک ارز
  Future<Either<Failure, ExchangeRate>> getCurrentRate(String currencyCode);

  /// ثبت نرخ جدید (Append-Only)
  Future<Either<Failure, ExchangeRate>> addRateEvent({
    required String currencyCode,
    required double rate,
    required RateSource source,
    double confidence = 1.0,
    String? notes,
  });

  /// دریافت تاریخچه نرخ‌ها
  Future<Either<Failure, List<ExchangeRate>>> getRateHistory({
    required String currencyCode,
    DateTime? startDate,
    DateTime? endDate,
    int? limit,
  });

  /// دریافت نرخ در یک تاریخ خاص (سفر در زمان)
  Future<Either<Failure, ExchangeRate>> getRateAtDate({
    required String currencyCode,
    required DateTime date,
  });

class AddExchangeRate implements UseCase<ExchangeRate, AddExchangeRateParams> {
  final ExchangeRateRepository repository;
  final EventLogger eventLogger;

  AddExchangeRate(this.repository, this.eventLogger);

  @override
  Future<Either<Failure, ExchangeRate>> call(AddExchangeRateParams params) async {
    // Validation
    if (params.rate <= 0) {
      return Left(ValidationFailure('نرخ ارز باید بزرگتر از صفر باشد'));
    }

    final result = await repository.addRateEvent(
      currencyCode: params.currencyCode,
      rate: params.rate,
      source: params.source,
      confidence: params.confidence,
      notes: params.notes,
    );

    // Log event for AI
    result.fold(
      (failure) => null,
      (rate) => eventLogger.log(
        type: EventType.exchangeRateUpdated,
        data: {
          'currencyCode': rate.currencyCode,
          'rate': rate.rate,
          'source': rate.source.name,
          'previousRate': params.previousRate,
        },
      ),
    );

    return result;
  }
}

          operationalProfit: operationalProfit,

// States
abstract class ExchangeRateState extends Equatable {
  const ExchangeRateState();
}

class ExchangeRateInitial extends ExchangeRateState {
  @override
  List<Object> get props => [];
}

class ExchangeRateLoading extends ExchangeRateState {
  @override
  List<Object> get props => [];
}

class ExchangeRateLoaded extends ExchangeRateState {
  final List<ExchangeRate> currentRates;
  final List<ExchangeRate>? history;
  final ExchangeRate? rateAtDate;

  const ExchangeRateLoaded({
    required this.currentRates,
    this.history,
    this.rateAtDate,
  });

  @override
  List<Object?> get props => [currentRates, history, rateAtDate];
}

class ExchangeRateError extends ExchangeRateState {
  final String message;

  const ExchangeRateError(this.message);

  @override
  List<Object> get props => [message];
}

// BLoC
class ExchangeRateBloc extends Bloc<ExchangeRateEvent, ExchangeRateState> {
  final GetCurrentRates getCurrentRates;
  final AddExchangeRate addExchangeRate;
  final GetRateHistory getRateHistory;
  final GetRateAtDateUseCase getRateAtDate;

  ExchangeRateBloc({
    required this.getCurrentRates,
    required this.addExchangeRate,
    required this.getRateHistory,
    required this.getRateAtDate,
  }) : super(ExchangeRateInitial()) {
    on<LoadCurrentRates>(_onLoadCurrentRates);
    on<AddNewRate>(_onAddNewRate);
    on<LoadRateHistory>(_onLoadRateHistory);
    on<GetRateAtDate>(_onGetRateAtDate);
  }

  Future<void> _onLoadCurrentRates(
    LoadCurrentRates event,
    Emitter<ExchangeRateState> emit,
  ) async {
    emit(ExchangeRateLoading());
    
    final result = await getCurrentRates(NoParams());
    
    result.fold(
      (failure) => emit(ExchangeRateError(failure.message)),
      (rates) => emit(ExchangeRateLoaded(currentRates: rates)),
    );
  }

  Future<void> _onAddNewRate(
    AddNewRate event,
    Emitter<ExchangeRateState> emit,
  ) async {
    // Get previous rate for logging
    double? previousRate;
    if (state is ExchangeRateLoaded) {
      final currentState = state as ExchangeRateLoaded;
      previousRate = currentState.currentRates
          .firstWhereOrNull((r) => r.currencyCode == event.currencyCode)
          ?.rate;
    }

    final result = await addExchangeRate(AddExchangeRateParams(
      currencyCode: event.currencyCode,
      rate: event.rate,
      source: event.source,
      notes: event.notes,
      previousRate: previousRate,
    ));

    result.fold(
      (failure) => emit(ExchangeRateError(failure.message)),
      (_) => add(LoadCurrentRates()), // Reload all rates
    );
  }

  Future<void> _onLoadRateHistory(
    LoadRateHistory event,
    Emitter<ExchangeRateState> emit,
  ) async {
    final currentState = state;
    if (currentState is! ExchangeRateLoaded) return;

    final result = await getRateHistory(GetRateHistoryParams(
      currencyCode: event.currencyCode,
      startDate: event.startDate,
      endDate: event.endDate,
    ));

    result.fold(
      (failure) => emit(ExchangeRateError(failure.message)),
      (history) => emit(ExchangeRateLoaded(
        currentRates: currentState.currentRates,
        history: history,
      )),
    );
  }

  Future<void> _onGetRateAtDate(
    GetRateAtDate event,
    Emitter<ExchangeRateState> emit,
  ) async {
    final currentState = state;
    if (currentState is! ExchangeRateLoaded) return;

    final result = await getRateAtDate(GetRateAtDateParams(
      currencyCode: event.currencyCode,
      date: event.date,
    ));

    result.fold(
      (failure) => emit(ExchangeRateError(failure.message)),
      (rate) => emit(ExchangeRateLoaded(
        currentRates: currentState.currentRates,
        history: currentState.history,
        rateAtDate: rate,
      )),
    );
  }
}

// ═══════════════════════════════════════════════════════════════
// lib/features/pricing/domain/services/pricing_service.dart

class PricingService {
  final PricingRepository pricingRepo;
  final ExchangeRateRepository exchangeRateRepo;

  PricingService({
    required this.pricingRepo,
    required this.exchangeRateRepo,
  });

  /// محاسبه بازه قیمت برای یک محصول
  Future<Either<Failure, PriceRange>> calculatePriceRange({
    required double costPrice,
    required String costCurrency,
  }) async {
    // دریافت تنظیمات
    final settingsResult = await pricingRepo.getSettings();
    
    return settingsResult.fold(
      (failure) => Left(failure),
      (settings) async {
        // تبدیل به ریال اگر ارزی است
        double costInIRR = costPrice;
        double exchangeRate = 1.0;

        if (costCurrency != 'IRR') {
          final rateResult = await exchangeRateRepo.getCurrentRate(costCurrency);
          
          return rateResult.fold(
            (failure) => Left(failure),
            (rate) {
              exchangeRate = rate.rate;
              costInIRR = costPrice * exchangeRate;

              return Right(_calculateRange(
                costInIRR: costInIRR,
                exchangeRate: exchangeRate,
                settings: settings,
              ));
            },
          );
        }

        return Right(_calculateRange(
          costInIRR: costInIRR,
          exchangeRate: exchangeRate,
          settings: settings,
        ));
      },
    );
  }

  PriceRange _calculateRange({
    required double costInIRR,
    required double exchangeRate,
    required PricingSettingsEntity settings,
  }) {
    return PriceRange(
      minPrice: costInIRR * (1 + settings.minProfitMargin),
      sellingPrice: costInIRR * (1 + settings.defaultProfitMargin),
      maxPrice: costInIRR * (1 + settings.maxProfitMargin),
      costPrice: costInIRR,
      exchangeRate: exchangeRate,
    );
  }
}

// ═══════════════════════════════════════════════════════════════
// lib/features/pricing/domain/services/profit_calculator.dart
// ═══════════════════════════════════════════════════════════════

class ProfitCalculator {
  /// سود دلاری - همیشه محاسبه میشه، ذخیره نمیشه
  double getProfitUSD(SalesItemEntity item) {
    if (item.exchangeRateAtSale == null || item.exchangeRateAtSale == 0) {
      return 0;
    }
    return item.profitIRR / item.exchangeRateAtSale!;
  }

  /// جمع سود دلاری تاریخی
  double getTotalHistoricalProfitUSD(List<SalesItemEntity> items) {
    return items.fold(0.0, (sum, item) => sum + getProfitUSD(item));
  }

  /// ارزش امروز (با نرخ فعلی)
  double getCurrentValueUSD(List<SalesItemEntity> items, double currentRate) {
    if (currentRate == 0) return 0;
    final totalIRR = items.fold(0.0, (sum, item) => sum + item.profitIRR);
    return totalIRR / currentRate;
  }

  /// نرخ مؤثر
  double getEffectiveRate(List<SalesItemEntity> items) {
    final totalIRR = items.fold(0.0, (sum, item) => sum + item.profitIRR);
    final totalUSD = getTotalHistoricalProfitUSD(items);
    if (totalUSD == 0) return 0;
    return totalIRR / totalUSD;
  }

  /// درصد کاهش ارزش
  double getValueLossPercent(List<SalesItemEntity> items, double currentRate) {
    final historical = getTotalHistoricalProfitUSD(items);
    if (historical == 0) return 0;
    final current = getCurrentValueUSD(items, currentRate);
    return ((historical - current) / historical) * 100;
  }

  /// تاثیر تغییر ارز
  FXRevaluationImpact getFXImpact(
    List<SalesItemEntity> items, 
    double currentRate,
  ) {
    return FXRevaluationImpact(
      historicalValue: getTotalHistoricalProfitUSD(items),
      currentValue: getCurrentValueUSD(items, currentRate),
    );
  }
}

// lib/core/database/migrations/migration_v2.dart
// ═══════════════════════════════════════════════════════════════

class MigrationV2 {
  static Future<void> migrate(Database db) async {
    await db.transaction(() async {
      // Step 1: Create new tables
      await _createExchangeRateEvents(db);
      await _createPricingSettings(db);
      
      // Step 2: Alter existing tables
      await _alterProducts(db);
      await _alterSalesItems(db);
      await _alterPurchaseItems(db);
      
      // Step 3: Initialize default data
      await _initializeDefaults(db);
      
      // Step 4: Migrate existing data
      await _migrateExistingData(db);
    });
  }

  static Future<void> _createExchangeRateEvents(Database db) async {
    await db.execute('''
      CREATE TABLE IF NOT EXISTS exchange_rate_events (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        currency_code TEXT NOT NULL,
        rate REAL NOT NULL,
        recorded_at TEXT NOT NULL,
        source TEXT NOT NULL DEFAULT 'manual',
        confidence REAL DEFAULT 1.0,
        notes TEXT,
        recorded_by_uuid TEXT NOT NULL
      )
    ''');

    await db.execute('''
      CREATE INDEX IF NOT EXISTS idx_exchange_rate_currency_date 
      ON exchange_rate_events(currency_code, recorded_at DESC)
    ''');
  }

  static Future<void> _createPricingSettings(Database db) async {
    await db.execute('''
      CREATE TABLE IF NOT EXISTS pricing_settings (
        id INTEGER PRIMARY KEY CHECK (id = 1),
        min_profit_margin REAL NOT NULL DEFAULT 0.20,
        default_profit_margin REAL NOT NULL DEFAULT 0.30,
        max_profit_margin REAL NOT NULL DEFAULT 0.50,
        base_currency TEXT NOT NULL DEFAULT 'IRR',
        track_currencies TEXT NOT NULL DEFAULT '["USD","EUR","AED"]',
        updated_at TEXT NOT NULL,
        updated_by_uuid TEXT NOT NULL
      )
    ''');
  }

  static Future<void> _alterProducts(Database db) async {
    // Check if columns exist before adding
    final columns = await db.rawQuery("PRAGMA table_info(products)");
    final columnNames = columns.map((c) => c['name'] as String).toSet();

    if (!columnNames.contains('min_price')) {
      await db.execute('ALTER TABLE products ADD COLUMN min_price REAL');
    }
    if (!columnNames.contains('max_price')) {
      await db.execute('ALTER TABLE products ADD COLUMN max_price REAL');
    }
    if (!columnNames.contains('base_currency_code')) {
      await db.execute(
        "ALTER TABLE products ADD COLUMN base_currency_code TEXT DEFAULT 'IRR'"
      );
    }
    if (!columnNames.contains('cost_exchange_rate')) {
      await db.execute(
        'ALTER TABLE products ADD COLUMN cost_exchange_rate REAL'
      );
    }
  }

  static Future<void> _alterSalesItems(Database db) async {
    final columns = await db.rawQuery("PRAGMA table_info(sales_items)");
    final columnNames = columns.map((c) => c['name'] as String).toSet();

    if (!columnNames.contains('exchange_rate_at_sale')) {
      await db.execute(
        'ALTER TABLE sales_items ADD COLUMN exchange_rate_at_sale REAL'
      );
    }
    if (!columnNames.contains('cost_exchange_rate')) {
      await db.execute(
        'ALTER TABLE sales_items ADD COLUMN cost_exchange_rate REAL'
      );
    }
  }

  static Future<void> _alterPurchaseItems(Database db) async {
    final columns = await db.rawQuery("PRAGMA table_info(purchase_items)");
    final columnNames = columns.map((c) => c['name'] as String).toSet();

    if (!columnNames.contains('currency_code')) {
      await db.execute(
        "ALTER TABLE purchase_items ADD COLUMN currency_code TEXT DEFAULT 'IRR'"
      );
    }
    if (!columnNames.contains('exchange_rate_at_purchase')) {
      await db.execute(
        'ALTER TABLE purchase_items ADD COLUMN exchange_rate_at_purchase REAL'
      );
    }
  }

  static Future<void> _initializeDefaults(Database db) async {
    // Insert default pricing settings
    await db.execute('''
      INSERT OR IGNORE INTO pricing_settings (
        id, min_profit_margin, default_profit_margin, max_profit_margin,
        base_currency, track_currencies, updated_at, updated_by_uuid
      ) VALUES (
        1, 0.20, 0.30, 0.50, 'IRR', '["USD","EUR","AED"]',
        datetime('now'), 'system'
      )
    ''');

    // Insert default USD rate if none exists
    final existingRates = await db.rawQuery(
      "SELECT COUNT(*) as count FROM exchange_rate_events WHERE currency_code = 'USD'"
    );
    
    if ((existingRates.first['count'] as int) == 0) {
      await db.execute('''
        INSERT INTO exchange_rate_events (
          currency_code, rate, recorded_at, source, confidence, recorded_by_uuid
        ) VALUES (
          'USD', 80000, datetime('now'), 'manual', 1.0, 'system'
        )
      ''');
    }
  }

  static Future<void> _migrateExistingData(Database db) async {
    // Update existing products with calculated min/max prices
    final settings = await db.rawQuery(
      'SELECT * FROM pricing_settings WHERE id = 1'
    );
    
    if (settings.isNotEmpty) {
      final minMargin = settings.first['min_profit_margin'] as double;
      final maxMargin = settings.first['max_profit_margin'] as double;

      await db.execute('''
        UPDATE products 
        SET 
          min_price = wac_price * (1 + $minMargin),
          max_price = wac_price * (1 + $maxMargin),
          base_currency_code = 'IRR'
        WHERE min_price IS NULL AND wac_price IS NOT NULL
      ''');
    }
  }
}

### ۱۰.۲ Migration Verification

```dart
// ═══════════════════════════════════════════════════════════════
// lib/core/database/migrations/migration_verifier.dart
// ═══════════════════════════════════════════════════════════════

class MigrationVerifier {
  final Database db;

  MigrationVerifier(this.db);

  Future<MigrationResult> verify() async {
    final errors = <String>[];
    final warnings = <String>[];

    // Check tables exist
    if (!await _tableExists('exchange_rate_events')) {
      errors.add('Table exchange_rate_events not found');
    }
    if (!await _tableExists('pricing_settings')) {
      errors.add('Table pricing_settings not found');
    }

    // Check columns in products
    final productColumns = await _getColumns('products');
    if (!productColumns.contains('min_price')) {
      errors.add('Column min_price not found in products');
    }
    if (!productColumns.contains('max_price')) {
      errors.add('Column max_price not found in products');
    }

    // Check default data
    final rateCount = await db.rawQuery(
      'SELECT COUNT(*) as count FROM exchange_rate_events'
    );
    if ((rateCount.first['count'] as int) == 0) {
      warnings.add('No exchange rates found - default should be created');
    }

    // Check pricing settings
    final settingsCount = await db.rawQuery(
      'SELECT COUNT(*) as count FROM pricing_settings'
    );
    if ((settingsCount.first['count'] as int) == 0) {
      errors.add('Pricing settings not initialized');
    }

    return MigrationResult(
      success: errors.isEmpty,
      errors: errors,
      warnings: warnings,
    );
  }

  Future<bool> _tableExists(String tableName) async {
    final result = await db.rawQuery(
      "SELECT name FROM sqlite_master WHERE type='table' AND name='$tableName'"
    );
    return result.isNotEmpty;
  }

  Future<Set<String>> _getColumns(String tableName) async {
    final columns = await db.rawQuery("PRAGMA table_info($tableName)");
    return columns.map((c) => c['name'] as String).toSet();
  }
}

class MigrationResult {
  final bool success;
  final List<String> errors;
  final List<String> warnings;

  MigrationResult({
    required this.success,
    required this.errors,
    required this.warnings,
  });
}
